# services/jwt_service.py
def create_access_token(payload: dict,
    secret_key: str, expires_minutes: int) -> str:
    to_encode = payload.copy()
    expire = datetime.utcnow() + timedelta(
        minutes=expires_minutes)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, secret_key,
                      algorithm="HS256")

# presentation/deps.py
ENVIRONMENT = os.getenv("ENVIRONMENT", "development")
SECRET_KEY  = os.getenv("SECRET_KEY", "dev-secret")

if ENVIRONMENT in ("prod","production") \
   and SECRET_KEY == "dev-secret":
    raise RuntimeError(
        "SECRET_KEY non securisee en production!")

def get_current_user(
    credentials = Depends(HTTPBearer()),
    db: Session = Depends(get_db),
) -> models.Utilisateur:
    payload = decode_access_token(
        credentials.credentials, SECRET_KEY)
    user = crud_user.get_user_by_email(
        db, payload["sub"])
    if user is None:
        raise HTTPException(401)
    return user

def require_admin(user=Depends(get_current_user)):
    if str(user.role) != "admin":
        raise HTTPException(403)
    return user

async def lookup_isbn(isbn: str):
