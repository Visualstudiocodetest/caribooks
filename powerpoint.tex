\begin{frame}{Stack technique et justifications}
  \begin{block}{Justifications des choix}
    \begin{itemize}\small
      \item \tech{FastAPI} : performances ASGI, Pydantic auto, Swagger natif \quad\badgeorange{Migration depuis Express.js}
      \item \tech{Next.js} : SSR pour le SEO vitrine, SPA pour le back-office
      \item \tech{MySQL} : relationnel éprouvé, transactions ACID
      \item \tech{SQLAlchemy} : requêtes paramétrées $\Rightarrow$ anti-injections SQL
      \item \tech{SQL vs NoSQL} : besoin fondamentalement relationnel (FK, contraintes d'intégrité stock/commande) $\Rightarrow$ pas de NoSQL au MVP ; \textbf{Redis} (cache clé-valeur) identifié comme accès NoSQL pertinent si la charge catalogue/sessions le justifie
      \item \tech{PostFinance} : TWINT natif, sandbox accessible, 100\% suisse \quad\badgeorange{Migration depuis Payrexx}
      \item \tech{GitHub Actions} : tests puis déploiement continu (SSH, systemd)
      \item \tech{Vercel} : déploiement frontend CDN global, gratuit
    \end{itemize}
  \end{block}
\end{frame}

    secret_key: str, expires_minutes: int) -> str:
    to_encode = payload.copy()
    expire = datetime.now(timezone.utc) + timedelta(
        minutes=expires_minutes)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, secret_key,
                      algorithm="HS256")

def get_current_user(
    credentials = Depends(security),
    db: Session = Depends(get_db),
) -> models.Utilisateur:
    token = credentials.credentials.strip()
    if token.lower().startswith("bearer "):
        token = token[7:].strip()
    payload = decode_access_token(token, SECRET_KEY)
    user = crud_user.get_user_by_email(
        db, payload["sub"])
    if user is None:
        raise HTTPException(401)
    return user

def require_admin(user=Depends(get_current_user)):
    if str(user.role) != "admin":
        raise HTTPException(403)
    return user
