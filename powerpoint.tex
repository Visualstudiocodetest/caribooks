\begin{frame}[fragile]{Authentification JWT --- Flux sécurisé}
  \begin{columns}[T]
    \column{0.53\textwidth}
\begin{lstlisting}[style=pysmall,caption={}]
# services/jwt_service.py
def create_access_token(payload: dict,
    secret_key: str, expires_minutes: int) -> str:
    to_encode = payload.copy()
    expire = datetime.utcnow() + timedelta(
        minutes=expires_minutes)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, secret_key,
                      algorithm="HS256")

# presentation/deps.py
ENVIRONMENT = os.getenv("ENVIRONMENT", "development")
SECRET_KEY  = os.getenv("SECRET_KEY", "dev-secret")

if ENVIRONMENT in ("prod","production") \
   and SECRET_KEY == "dev-secret":
    raise RuntimeError(
        "SECRET_KEY non securisee en production!")

def get_current_user(
    credentials = Depends(HTTPBearer()),
    db: Session = Depends(get_db),
) -> models.Utilisateur:
    payload = decode_access_token(
        credentials.credentials, SECRET_KEY)
    user = crud_user.get_user_by_email(
        db, payload["sub"])
    if user is None:
        raise HTTPException(401)
    return user

def require_admin(user=Depends(get_current_user)):
    if str(user.role) != "admin":
        raise HTTPException(403)
    return user
\end{lstlisting}
    \column{0.44\textwidth}
    \begin{block}{Flux d'authentification}
      \begin{enumerate}\small
        \item \texttt{POST /auth/token} (email + mdp)
        \item Vérification hash \textbf{bcrypt} (passlib)
        \item Génération \textbf{JWT HS256}\\
          \texttt{\{sub, role, exp: 480 min\}}
        \item Stockage client → \texttt{Bearer token}
        \item Middleware \texttt{get\_current\_user} sur chaque route protégée
        \item \texttt{require\_admin} pour le back-office
      \end{enumerate}
    \end{block}
    \vspace{5pt}
    \begin{tikzpicture}

# presentation/book_router.py
@router.get("/isbn/{isbn}")
async def lookup_isbn(isbn: str):
    async with httpx.AsyncClient() as client:

        # 1. OpenLibrary (source principale)
        resp = await client.get(
            "https://openlibrary.org/api/books",
            params={"bibkeys": f"ISBN:{isbn}",
                    "format": "json",
                    "jscmd":  "data"})
        data = resp.json()
        if f"ISBN:{isbn}" in data:
            book = data[f"ISBN:{isbn}"]
            return {"isbn":    isbn,
                "titre":   book.get("title"),
                "auteur":  book["authors"][0]["name"],
                "editeur": book["publishers"][0]["name"],
                "source":  "openlibrary"}

        # 2. Fallback Google Books
        resp2 = await client.get(
            "https://www.googleapis.com/books/v1/volumes",
            params={"q": f"isbn:{isbn}"})
        if resp2.json().get("totalItems", 0) > 0:
            info = resp2.json()["items"][0]["volumeInfo"]
            return {"titre":  info.get("title"),
                    "source": "google_books", ...}

    raise HTTPException(404,
        "ISBN non trouve. Saisie manuelle.")
\end{lstlisting}
